cmake_minimum_required(VERSION 3.5)
project(elastic_tracker)
set(CMAKE_CXX_STANDARD 17)
find_package(Eigen3 REQUIRED)
find_package(catkin REQUIRED COMPONENTS roscpp sensor_msgs nav_msgs quadrotor_msgs planner_execution)
catkin_package(INCLUDE_DIRS include LIBRARIES elastic_tracker CATKIN_DEPENDS roscpp sensor_msgs nav_msgs quadrotor_msgs planner_execution DEPENDS Eigen3)
add_library(elastic_tracker src/traj_opt.cpp src/tracking.cpp src/tracking_ros.cpp)
target_include_directories(elastic_tracker PUBLIC include ${EIGEN3_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
target_link_libraries(elastic_tracker ${catkin_LIBRARIES})
add_dependencies(elastic_tracker ${catkin_EXPORTED_TARGETS})
install(TARGETS elastic_tracker LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
