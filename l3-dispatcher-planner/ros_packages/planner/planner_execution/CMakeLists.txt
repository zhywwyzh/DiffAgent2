cmake_minimum_required(VERSION 3.5)
project(planner_execution)
set(CMAKE_CXX_STANDARD 17)
find_package(catkin REQUIRED COMPONENTS roscpp actionlib quadrotor_msgs nav_msgs std_msgs geometry_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES planner_execution CATKIN_DEPENDS roscpp actionlib quadrotor_msgs nav_msgs std_msgs geometry_msgs)
add_library(planner_execution src/execution.cpp src/settle_monitor.cpp src/action_server_ros.cpp src/command_output_ros.cpp)
target_include_directories(planner_execution PUBLIC include ${catkin_INCLUDE_DIRS})
target_link_libraries(planner_execution ${catkin_LIBRARIES})
add_dependencies(planner_execution ${catkin_EXPORTED_TARGETS})
install(TARGETS planner_execution ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
